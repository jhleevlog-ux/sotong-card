        function formatDate(dateObj) {

            const date = new Date(dateObj);

            const yyyy = date.getFullYear();

            const mm = String(date.getMonth() + 1).padStart(2, '0');

            const mi = String(date.getMinutes()).padStart(2, '0');

            const mm = String(date.getMonth() + 1).padStart(2, '0');

            const dd = String(date.getDate()).padStart(2, '0');

            const hh = String(date.getHours()).padStart(2, '0');

            const mi = String(date.getMinutes()).padStart(2, '0');

            return `${yyyy}-${mm}-${dd}T${hh}:${mi}`;

        }


        function loadData() {

            }

        }

        }

        function renderStats() {

            const now = new Date();

            const currentMonthStr = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;

            const thisMonthLogs = logs.filter(item => item.checkoutTime && item.checkoutTime.startsWith(currentMonthStr));

            document.getElementById('statTotalCount').textContent = `${thisMonthLogs.length}회`;


            const statCurrentStateEl = document.getElementById('statCurrentState');

            if (currentCheckoutItem) {

                statCurrentStateEl.className = 'text-lg font-bold text-emerald-600';

            document.getElementById('statLastUser').textContent = logs[0] ? logs[0].borrower : '-';

        }


        function renderTable() {

            const query = searchInput.value.trim().toLowerCase();

            const filteredLogs = logs.filter(item => 

                (item.borrower && item.borrower.toLowerCase().includes(query)) || 

                (item.purpose && item.purpose.toLowerCase().includes(query))

            );


            historyTableBody.innerHTML = '';

            if (filteredLogs.length === 0) {

                emptyState.classList.remove('hidden');

                return;

            }

            emptyState.classList.add('hidden');


            filteredLogs.forEach(item => {

                const tr = document.createElement('tr');

                tr.className = 'hover:bg-slate-50 transition-colors';

                const isCheckedOut = item.status === 'CHECKED_OUT';

                const isOverdue = isCheckedOut && new Date() > new Date(item.expectedReturnTime);


                let statusBadge = isCheckedOut 

                    ? (isOverdue ? `<span class="px-2.5 py-1 text-xs font-bold bg-rose-100 text-rose-700 rounded-full">반납지연</span>` : `<span class="px-2.5 py-1 text-xs font-bold bg-amber-100 text-amber-700 rounded-full">사용중</span>`)

                    : `<span class="px-2.5 py-1 text-xs font-semibold bg-emerald-100 text-emerald-700 rounded-full">반납완료</span>`;


                tr.innerHTML = `

                    <td class="px-4 py-3.5 whitespace-nowrap">${statusBadge}</td>

                    <td class="px-4 py-3.5 font-bold text-slate-800 whitespace-nowrap">${item.borrower || ''}</td>

                    <td class="px-4 py-3.5 font-mono text-xs text-slate-600 whitespace-nowrap">${item.checkoutTime || ''}</td>

                    <td class="px-4 py-3.5 font-mono text-xs font-semibold ${isOverdue ? 'text-rose-600' : 'text-slate-700'} whitespace-nowrap">${item.expectedReturnTime || ''}</td>

                    <td class="px-4 py-3.5 font-mono text-xs text-slate-600 whitespace-nowrap">${item.actualReturnTime || '-'}</td>

                    <td class="px-4 py-3.5 text-xs text-slate-700 max-w-xs truncate">${item.purpose || ''}</td>

                    <td class="px-4 py-3.5 text-right whitespace-nowrap">

                        ${isCheckedOut ? `<button onclick="quickReturnItem('${item.id}')" class="px-2.5 py-1 bg-amber-50 text-amber-700 border border-amber-200 rounded-lg text-xs font-semibold">반납하기</button>` : `<button onclick="deleteItem('${item.id}')" class="text-slate-300 hover:text-rose-500 px-2 py-1"><i class="fa-regular fa-trash-can"></i></button>`}

                    </td>

                `;

                historyTableBody.appendChild(tr);

            });

        }

        function openCheckoutModal() {

            displayCheckoutTime.value = formatDate(new Date());

        }


        function closeCheckoutModal() { checkoutModal.classList.add('hidden'); }

        function openReturnModal() {

            if (!currentCheckoutItem) return;

            document.getElementById('displayReturnTime').value = formatDate(new Date());

            returnModal.classList.remove('hidden');

        }

        function closeReturnModal() { returnModal.classList.add('hidden'); }
